cmake_minimum_required(VERSION 3.20)
project(TrackPadCAD LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(gesture INTERFACE)
target_include_directories(gesture INTERFACE include)
add_executable(trackpad-cad src/main.cpp)
target_link_libraries(trackpad-cad PRIVATE gesture)
if(WIN32)
  target_sources(trackpad-cad PRIVATE src/windows.cpp)
  target_link_libraries(trackpad-cad PRIVATE hid user32)
  target_compile_definitions(trackpad-cad PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MINGW)
  target_link_options(trackpad-cad PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(gesture-tests tests/gesture_tests.cpp)
target_link_libraries(gesture-tests PRIVATE gesture)
if(MINGW)
  target_link_options(gesture-tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME gesture-tests COMMAND gesture-tests)
