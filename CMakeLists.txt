cmake_minimum_required(VERSION 3.20)
project(TrackPadCAD LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(gesture INTERFACE)
target_include_directories(gesture INTERFACE include)
add_executable(trackpad-cad src/main.cpp)
target_link_libraries(trackpad-cad PRIVATE gesture)
if(WIN32)
  target_sources(trackpad-cad PRIVATE src/windows.cpp)
  target_link_libraries(trackpad-cad PRIVATE hid user32)
  target_compile_definitions(trackpad-cad PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  find_program(PWSH_EXECUTABLE pwsh REQUIRED)
  set(TRACKPAD_ICON "${CMAKE_BINARY_DIR}/assets/trackpad-cad.ico")
  add_custom_command(OUTPUT "${TRACKPAD_ICON}"
    COMMAND "${PWSH_EXECUTABLE}" -NoProfile -File "${CMAKE_SOURCE_DIR}/tools/make-icon.ps1" -Output "${TRACKPAD_ICON}"
    DEPENDS "${CMAKE_SOURCE_DIR}/tools/make-icon.ps1")
  add_custom_target(trackpad-icon DEPENDS "${TRACKPAD_ICON}")
  configure_file("${CMAKE_SOURCE_DIR}/src/launcher.rc.in" "${CMAKE_BINARY_DIR}/launcher.rc" @ONLY)
  set_source_files_properties("${CMAKE_BINARY_DIR}/launcher.rc" PROPERTIES OBJECT_DEPENDS
    "${CMAKE_SOURCE_DIR}/run-trackpad.ps1;${CMAKE_SOURCE_DIR}/src/HidMaestroBridge.cs;${CMAKE_SOURCE_DIR}/build/hidmaestro/HIDMaestro.Core.dll;${CMAKE_SOURCE_DIR}/experiments/hidmaestro/spacemouse-pro.hex;${CMAKE_SOURCE_DIR}/TrackPad CAD.json;${CMAKE_SOURCE_DIR}/browser-extension/main-hook.js;${CMAKE_SOURCE_DIR}/browser-extension/relay.js;${CMAKE_SOURCE_DIR}/browser-extension/manifest.json")
  add_executable(trackpad-cad-launcher WIN32 src/launcher.cpp src/main.cpp src/windows.cpp "${CMAKE_BINARY_DIR}/launcher.rc")
  add_dependencies(trackpad-cad-launcher trackpad-icon)
  target_link_libraries(trackpad-cad-launcher PRIVATE gesture hid user32 shell32 ole32)
  target_compile_definitions(trackpad-cad-launcher PRIVATE TPC_CAPTURE_LIBRARY NOMINMAX WIN32_LEAN_AND_MEAN)
  set_target_properties(trackpad-cad-launcher PROPERTIES OUTPUT_NAME "TrackPad CAD" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}")
  add_custom_command(TARGET trackpad-cad-launcher POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:trackpad-cad-launcher>" "${CMAKE_SOURCE_DIR}/TrackPad CAD Setup.exe")
endif()
if(MINGW)
  target_link_options(trackpad-cad PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(trackpad-cad-launcher PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(gesture-tests tests/gesture_tests.cpp)
target_link_libraries(gesture-tests PRIVATE gesture)
if(MINGW)
  target_link_options(gesture-tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME gesture-tests COMMAND gesture-tests)
